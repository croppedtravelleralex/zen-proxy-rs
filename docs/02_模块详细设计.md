## 一、设计总则

### 1.1 榫卯结构定义

```
每个模块必须满足：
  ① trait 定义文件头部（接口层）
  ② struct 实现放文件主体（实现层，内部可任意紧耦合）
  ③ 禁止跨模块直接操作 struct 字段或内部方法
  ④ 模块间通信必须通过 trait 方法

文件内部分段规范：
  // ── 接口层 ──
  pub trait Xxx: Send + Sync {
      fn method_a(&self);
      fn method_b(&self) -> usize;
  }

  // ── 实现层 ──
  pub struct XxxImpl { ... }

  impl Xxx for XxxImpl {
      fn method_a(&self) { ... }  // 内部可以任意紧耦合
  }
```

### 1.2 模块依赖规则

```
依赖方向：只允许单向
  proxy.rs -> PoolManager -> 各 Pool
  proxy.rs -> DataCollector
  proxy.rs -> Sanitizer
  PoolManager -> UpstreamHealth
  PoolManager -> NodeProvider (通过 DispatchPool 泛型参数)
  server/admin.rs -> DefaultCollector (查询方法)
  runtime -> DefaultCollector (定时任务: WAL滚动/聚合快照)

禁止反向依赖：
  各 Pool -> proxy.rs          ❌
  各 Pool -> 其他 Pool        ❌ (由 PoolManager 编排)
  DataCollector -> 各池内部   ❌ (只收事件，不读取)
  AuthGate -> server.rs       ❌ (只接 AdminRequest)
  Sanitizer -> 业务模块内部   ❌ (只处理字符串/标签/查询边界)
```

## 二、PoolManager 模块

### ── 接口层 ──

```rust
/// 池管理器榫头
///
/// proxy.rs 只认这个 trait，不认具体实现类型。
/// 想换池组实现，在 main.rs 换类型参数即可。
pub trait PoolManager: Send + Sync {
    /// 为一次请求调度一个可用节点
    ///
    /// 返回:
    ///   Ok(DispatchResult) — 包含节点引用和 SOCKS5 客户端
    ///   Err(DispatchError::NoResource) — 调度池空，-999
    fn dispatch(&self, req: &RequestMeta) -> Result<DispatchResult, DispatchError>;

    /// 汇报一次请求的结果
    ///
    /// PoolManager 根据结果决定节点去向：
    ///   Success     -> 留在/回调度池
    ///   RateLimited -> 进 429 池
    ///   Error       -> 进探活期
    fn report(&self, node_id: NodeId, result: ResultKind, latency_us: u64);

    /// 当前各池统计快照
    fn pool_stats(&self) -> PoolStats;

    /// 手动熔断：将所有节点移入死池
    fn fuse_all(&self);

    /// 手动恢复：将所有节点移回调度池
    fn unfuse_all(&self);
}

/// 调度结果
pub struct DispatchResult {
    pub node: NodeRef,
    pub client: reqwest::Client,  // 已配置好 SOCKS5 的 client
}

/// 调度错误
pub enum DispatchError {
    /// 所有节点不可用（调度池空 + 直连 fallback 关闭）
    NoResource,
}

/// 请求元数据
pub struct RequestMeta {
    pub model: String,
    pub stream: bool,
    pub body_size: u64,
}

/// 使用结果
pub enum ResultKind {
    Success(u16),              // HTTP 200
    RateLimited,               // HTTP 429
    Error { kind: ErrorKind },  // 其他错误
}

pub enum ErrorKind {
    Timeout,
    ConnectionRefused,
    DnsFailure,
    SocksHandshake,
    Upstream5xx,
    Other(String),
}
```

### ── 实现层 ──

```rust
/// PoolManager 默认实现
///
/// 内部紧耦合——四个池的状态机和流转逻辑全部在这里。
/// 但对外只露 trait 定义的方法，外部不感知内部实现。
pub struct PoolManagerImpl<D, A, R, K, C>
where
    D: Pool,
    A: ActivePool,
    R: RateLimitedPool,
    K: DeadPool,
    C: DataCollector,
{
    dispatch: D,
    active: Arc<A>,
    ratelimited: R,
    dead: K,
    collector: C,
    health: UpstreamHealth,    // 保留原背压监控
    fuse: AtomicBool,          // 熔断开关
}

impl<D, A, R, K, C> PoolManager for PoolManagerImpl<D, A, R, K, C>
where
    D: Pool<NodeId = NodeId>,
    A: ActivePool<NodeId = NodeId>,
    R: RateLimitedPool<NodeId = NodeId>,
    K: DeadPool<NodeId = NodeId>,
    C: DataCollector,
{
    fn dispatch(&self, req: &RequestMeta) -> Result<DispatchResult, DispatchError> {
        // 1. 熔断检查
        if self.fuse.load(Ordering::Acquire) {
            return Err(DispatchError::NoResource);
        }

        // 2. 从调度池选节点
        let node = self.dispatch.acquire()
            .ok_or(DispatchError::NoResource)?;

        // 3. 通知使用池
        self.active.register(&node);

        // 4. 获取 SOCKS5 客户端（由 SessionPool 管理）
        let client = self.dispatch.get_client(&node)?;

        // 5. 统计
        self.collector.record_pool(PoolEvent {
            pool: PoolId::Dispatch,
            action: PoolAction::Acquire,
            node_id: Some(node.id()),
            ..Default::default()
        });

        Ok(DispatchResult { node, client })
    }

    fn report(&self, node_id: NodeId, result: ResultKind, latency_us: u64) {
        match &result {
            ResultKind::Success(_) => {
                self.dispatch.release(&node_id, &result);
                self.active.release(&node_id, &result);
            }
            ResultKind::RateLimited => {
                self.dispatch.remove(&node_id);        // 从调度池移除
                self.active.release(&node_id, &result); // 从使用池释放
                self.ratelimited.quarantine(node_id);   // 进入429池
            }
            ResultKind::Error { .. } => {
                self.dispatch.remove(&node_id);
                self.active.release(&node_id, &result);
                // 启动异步探活期
                let _ = tokio::spawn(probe_period(
                    node_id,
                    Arc::clone(&self.dispatch),
                    Arc::clone(&self.dead),
                    Arc::clone(&self.collector),
                ));
            }
        }
        self.health.record(result.http_status());
        self.collector.record_request(RequestEvent {
            status_code: result.http_status(),
            latency_us,
            ..Default::default()
        });
    }
}
```

## 三、调度池 DispatchPool

### ── 接口层 ──

```rust
/// 通用池接口
pub trait Pool: Send + Sync {
    type NodeId: Clone + Hash + Eq;

    /// 获取一个可用节点
    fn acquire(&self) -> Option<NodeRef<Self::NodeId>>;

    /// 回收节点
    fn release(&self, node_id: &Self::NodeId, result: &ResultKind);

    /// 移除节点
    fn remove(&self, node_id: &Self::NodeId);

    /// 添加节点
    fn add(&self, node: NodeRef<Self::NodeId>);

    /// 当前可用节点数
    fn available(&self) -> usize;

    /// 池名称
    fn name(&self) -> &'static str;
}
```

### ── 实现层 ──

```rust
/// 调度池——加权随机选人
pub struct DispatchPool<P: NodeProvider> {
    nodes: RwLock<Vec<PoolNode<P>>>,
    next_probe: AtomicU64,          // 下次探活时间
    session_cache: RwLock<HashMap<P::NodeId, Client>>,  // SOCKS5 client 缓存
}

struct PoolNode<P> {
    provider: P,
    id: P::NodeId,
    base_score: AtomicU64,              // 健康度 0~100 (f64 bits)
    consecutive_successes: AtomicU32,
    recent_results: ArrayVec<bool, 20>, // 最近20次结果（环形缓冲）
    avg_latency_ms: AtomicU64,
    idle_since: AtomicI64,              // unix_ms, -1 = 使用中
    max_concurrent: AtomicU32,
}

impl<P: NodeProvider> Pool for DispatchPool<P> {
    fn acquire(&self) -> Option<NodeRef<P::NodeId>> {
        let now = unix_ms();
        let nodes = self.nodes.read();
        let total: f64 = nodes.iter()
            .filter(|n| n.is_available(now))
            .map(|n| n.score(now))
            .sum();

        if total <= 0.0 { return None; }

        let pick = fastrand::f64() * total;
        let mut cumulative = 0.0;
        for node in nodes.iter() {
            if !node.is_available(now) { continue; }
            cumulative += node.score(now);
            if pick <= cumulative {
                node.idle_since.store(-1, Ordering::Relaxed);
                return Some(NodeRef { id: node.id.clone() });
            }
        }
        None  // 不应达到
    }

    fn release(&self, node_id: &P::NodeId, result: &ResultKind) {
        let nodes = self.nodes.read();
        if let Some(node) = nodes.iter().find(|n| n.id == *node_id) {
            match result {
                ResultKind::Success(_) => {
                    let mut current = node.base_score.load(Ordering::Relaxed);
                    loop {
                        let current_f = f64::from_bits(current);
                        let new_f = current_f + 3.0;
                        match node.base_score.compare_exchange_weak(
                            current, f64::to_bits(new_f),
                            Ordering::Relaxed, Ordering::Relaxed,
                        ) {
                            Ok(_) => break,
                            Err(v) => current = v,
                        }
                    }
                    node.consecutive_successes.fetch_add(1, Ordering::Relaxed);
                }
                _ => {
                    node.consecutive_successes.store(0, Ordering::Relaxed);
                }
            }
            node.idle_since.store(unix_ms(), Ordering::Relaxed);
        }
    }
}

### 调度核心：健康度评分算法

```rust
impl<P: NodeProvider> PoolNode<P> {
    /// 综合评分（调度池选人用）
    fn score(&self, now: i64) -> f64 {
        let health = f64::from_bits(self.base_score.load(Ordering::Relaxed)) / 100.0;        // 0~1
        let success = self.success_rate() * 0.5;                                             // 0~0.5
        let idle = ((now - self.idle_since.load(Ordering::Relaxed)) as f64                   // 0~1
                    / 120_000.0).clamp(0.0, 1.0) * 0.15;
        let latency = 1.0 - (f64::from_bits(self.avg_latency_ms.load(Ordering::Relaxed))    // 0~1
                    / 10_000.0).clamp(0.0, 1.0) * 0.10;
        let momentum = (self.consecutive_successes.load(Ordering::Relaxed)                   // 0~1
                    as f64 / 50.0).clamp(0.0, 1.0) * 0.05;

        health * 0.50 + success / 0.5 * 0.20 + idle * 0.15 + latency * 0.10 + momentum * 0.05
    }
}
```

## 四、使用池 ActivePool

### ── 接口层 ──

```rust
/// 使用池——管理与释放活跃连接
pub trait ActivePool: Send + Sync {
    type NodeId: Clone + Hash + Eq;

    /// 注册一个节点（开始使用）
    fn register(&self, node: &NodeRef<Self::NodeId>);
    /// 释放一个节点（使用完毕）
    fn release(&self, node_id: &Self::NodeId, result: &ResultKind);
    /// 当前活跃请求总数
    fn active_count(&self) -> usize;
    /// 每个节点的当前并发数
    fn per_node_concurrency(&self) -> HashMap<Self::NodeId, i64>;
}
```

### ── 实现层 ──

```rust
pub struct ActivePoolImpl<NodeId: Hash + Eq> {
    entries: RwLock<HashMap<NodeId, ActiveEntry>>,
}

struct ActiveEntry {
    active_requests: AtomicI64,    // 当前并发请求数
    max_concurrent: AtomicU32,    // 动态上限
    entered_at: Instant,
    last_used: AtomicI64,         // unix_ms
}

impl<NodeId: Hash + Eq + Clone> ActivePool for ActivePoolImpl<NodeId> {
    type NodeId = NodeId;

    fn register(&self, node: &NodeRef<NodeId>) {
        let mut entries = self.entries.write();
        let entry = entries.entry(node.id.clone())
            .or_insert_with(|| ActiveEntry {
                max_concurrent: AtomicU32::new(5),   // 初始并发上限5
                entered_at: Instant::now(),
                ..Default::default()
            });
        entry.active_requests.fetch_add(1, Ordering::Relaxed);
    }

    fn release(&self, node_id: &NodeId, result: &ResultKind) {
        if let Some(entry) = self.entries.read().get(node_id) {
            let prev = entry.active_requests.fetch_sub(1, Ordering::Relaxed);
            match result {
                ResultKind::Success(_) => {
                    // 成功 -> max_concurrent 缓慢上调
                    let mc = entry.max_concurrent.load(Ordering::Relaxed);
                    if prev <= mc as i64 {
                        entry.max_concurrent
                            .fetch_update(Ordering::Relaxed, Ordering::Relaxed,
                                |x| Some((x + 1).min(20)))
                            .ok();
                    }
                }
                ResultKind::RateLimited | ResultKind::Error { .. } => {
                    // 失败 -> max_concurrent 砍半
                    entry.max_concurrent
                        .fetch_update(Ordering::Relaxed, Ordering::Relaxed,
                            |x| Some((x / 2).max(1)))
                        .ok();
                }
            }
            entry.last_used.store(unix_ms(), Ordering::Relaxed);
        }
    }

    fn active_count(&self) -> usize {
        self.entries.read().len()
    }

    fn per_node_concurrency(&self) -> HashMap<NodeId, i64> {
        self.entries.read().iter()
            .map(|(id, e)| (id.clone(), e.active_requests.load(Ordering::Relaxed)))
            .collect()
    }
}
```

## 五、429池 RateLimitedPool

### ── 接口层 ──

```rust
/// 429池额外能力（探活调度用）
pub trait RateLimitedPool: Pool {
    /// 记录一个节点被 429 限流
    fn quarantine(&self, node_id: Self::NodeId);
    /// 选择一批节点做探活（次日起，分批抽取）
    fn select_for_probe(&self, batch_size: usize) -> Vec<Self::NodeId>;
    /// 探活成功，恢复节点
    fn recover(&self, node_id: &Self::NodeId);
    /// 当天隔离的节点数
    fn quarantined_today(&self) -> usize;
}
```

### ── 实现层 ──

```rust
pub struct RateLimitedPoolImpl<NodeId: Hash + Eq> {
    entries: RwLock<HashMap<NodeId, QuarantineEntry>>,
}

struct QuarantineEntry {
    /// 上次429的日期 (yyyyMMdd 格式)
    last_429_date: u32,
    /// 连续几天被限流
    consecutive_days: u32,
}

impl<NodeId: Hash + Eq + Clone> RateLimitedPool for RateLimitedPoolImpl<NodeId> {
    fn quarantine(&self, node_id: NodeId) {
        let today = today_ymd();
        let mut entries = self.entries.write();
        let entry = entries.entry(node_id)
            .or_insert(QuarantineEntry { last_429_date: today, consecutive_days: 1 });

        if entry.last_429_date == today {
            entry.consecutive_days += 1;
        } else if entry.last_429_date < today - 1 {
            entry.consecutive_days = 1;
        }
        entry.last_429_date = today;
    }

    fn select_for_probe(&self, batch_size: usize) -> Vec<NodeId> {
        let today = today_ymd();
        let entries = self.entries.read();

        // 只选今天之前的节点（当天被429的不碰）
        let candidates: Vec<_> = entries.iter()
            .filter(|(_, e)| e.last_429_date < today)
            .map(|(id, _)| id.clone())
            .collect();

        // 随机抽取 batch_size 个
        fastrand::choose_multiple(candidates, batch_size.min(candidates.len()))
    }
}
```

### 探活规则

```
① 节点进 429 池 -> 记录 last_429_date = today
② 当天不碰它（不探活、不解禁、不调度）
③ 次日8点开始探活周期:
   每 2~4 小时（随机间隔），从429池抽 3~5 个节点
   用真实 HTTP 请求走完整 SOCKS5 链路
   探活成功(200) -> 回调度池，健康度=50
   探活失败(429) -> 留池，等下一轮
   连续 30 天 429 -> 通知管理员
④ 当日429池为空 -> 停止探活，等明天
```

## 六、死池 DeadPool

### ── 接口层 ──

```rust
pub trait DeadPool: Pool {
    /// 节点进入死池
    fn bury(&self, node_id: Self::NodeId);
    /// 选择所有节点做每日探活
    fn select_all_for_probe(&self) -> Vec<Self::NodeId>;
    /// 探活成功，恢复节点
    fn recover(&self, node_id: &Self::NodeId);
    /// 累计死亡次数
    fn dead_count(&self, node_id: &Self::NodeId) -> u32;
}
```

### ── 实现层 ──

```rust
pub struct DeadPoolImpl<NodeId: Hash + Eq> {
    entries: RwLock<HashMap<NodeId, DeadEntry>>,
    collector: Arc<dyn DataCollector>,
}

struct DeadEntry {
    entered_at: Instant,
    dead_count: u32,
    last_probe_ok: Option<bool>,
}

impl<NodeId: Hash + Eq + Clone> DeadPool for DeadPoolImpl<NodeId> {
    fn bury(&self, node_id: NodeId) {
        let mut entries = self.entries.write();
        let entry = entries.entry(node_id)
            .or_insert(DeadEntry {
                entered_at: Instant::now(),
                dead_count: 0,
                last_probe_ok: None,
            });
        entry.dead_count += 1;
    }

    fn select_all_for_probe(&self) -> Vec<NodeId> {
        self.entries.read().keys().cloned().collect()
    }

    fn recover(&self, node_id: &NodeId) {
        self.entries.write().remove(node_id);
    }
}

/// 探活期异步任务（非独立池，瞬态）
async fn probe_period<NodeId: Clone + Hash + Eq>(
    node_id: NodeId,
    dispatch: Arc<dyn Pool<NodeId = NodeId>>,
    dead: Arc<dyn DeadPool<NodeId = NodeId>>,
    collector: Arc<dyn DataCollector>,
) {
    // 3轮探测，间隔500ms
    let mut success_count = 0u32;
    for _ in 0..3 {
        if probe_node_once(&node_id).await {
            success_count += 1;
        }
        tokio::time::sleep(Duration::from_millis(500)).await;
    }

    collector.record_probe(ProbeEvent {
        node_id: format!("{:?}", node_id),
        pool: PoolId::ProbePeriod,
        success_count,
        total_count: 3,
    });

    if success_count >= 1 {
        // 有成功 -> 网络波动
        // 等5分钟观察
        tokio::time::sleep(Duration::from_secs(300)).await;
        if !probe_node_once(&node_id).await {
            // 5分钟内又失败 -> 重新探活期
            return probe_period(node_id, dispatch, dead, collector).await;
        }
        // 恢复
        dispatch.add(NodeRef { id: node_id });
    } else {
        // 3轮全败 -> 死池
        dead.bury(node_id);
    }
}
```

## 七、探活期 ProbePeriod

> 不是独立池，是 tokio::spawn 异步任务。
> 见死池实现层中的 `probe_period()` 函数。

```

### ── RingBuffer (环形缓冲区 + WAL) ──

```rust
/// 环形缓冲区——固定内存、零分配、WAL 持久化、崩溃回放
pub struct RingBuffer<T> {
    buffer: Box<[UnsafeCell<MaybeUninit<T>>]>,  // 预先分配, 固定 10k 槽位
    capacity: usize,                             // 固定 10000
    head: AtomicU64,                              // 写位置, 单调递增, 用 % capacity wrap
    write_lock: Mutex<()>,                       // 写同步锁, 防止 wrap-around 时读写冲突
    wal: Mutex<BufWriter<File>>,                 // 追加写 WAL 文件
    wal_path: PathBuf,                           // telemetry.wal
}

impl<T: Clone + Serialize> RingBuffer<T> {
    /// 写入: 内存 O(1) + WAL 追加, 共 ~0.8μs
    fn push(&self, item: &T) {
        let _lock = self.write_lock.lock().unwrap();
        let idx = self.head.fetch_add(1, Ordering::Relaxed) % self.capacity as u64;
        // 直接 memcpy 到预先分配的槽位, 零堆分配
        unsafe { *self.buffer[idx as usize].get() = MaybeUninit::new(item.clone()); }
        // WAL 追加写
        let mut wal = self.wal.lock();
        if let Err(e) = serde_json::to_writer(&mut *wal, item) {
            tracing::warn!("WAL serialization error: {}", e);
        }
        if let Err(e) = wal.write_all(b"\n") {
            tracing::warn!("WAL write error: {}", e);
        }
        // 不每条 fsync——VPS 云盘 fsync 耗时 1-10ms, 1800 req/s 会 IO 崩塌
        // 由后台定时任务每 100ms 批量 fsync 一次
        // 崩溃最多丢 ~180 条 (100ms 窗口)
    }

    /// 查询: 从 head-1 向 tail 扫描, 匹配 filter, 返回 cursor 分页结果
    /// 复杂度 O(N), N=当前存活条数, 10k 条 ~2ms
    fn query(
        &self,
        since: i64,
        filters: &[Filter],
        limit: usize,
        cursor: Option<u64>,   // 上次返回的 rid 对应的顺序号
    ) -> (Vec<T>, Option<u64>, bool) {
        let head = self.head.load(Ordering::Relaxed);
        let start = cursor.map(|c| c + 1).unwrap_or(0).max(head.saturating_sub(self.capacity as u64));
        // 从 start -> head 正向扫描 (先入先出)
        // 匹配 since + filters
        // 返回最多 limit 条 + 新 cursor (当前最后一条的顺序号) + truncated 标记
    }

    /// 启动时回放 WAL
    fn replay(&mut self, wal_path: &Path) {
        let file = File::open(wal_path).ok();
        if let Some(f) = file {
            let reader = BufReader::new(f);
            for line in reader.lines() {
                if let Ok(l) = line {
                    if let Ok(item) = serde_json::from_str::<T>(&l) {
                        let idx = self.head.fetch_add(1, Ordering::Relaxed) % self.capacity as u64;
                        unsafe { *self.buffer[idx as usize].get() = MaybeUninit::new(item); }
                    }
                }
            }
        }
        // 如果 WAL 行数 > capacity, 只保留最近 capacity 条
    }
}

/// 内存布局: 10k × ~200 bytes = ~2MB 固定内存, 运行时零分配
/// WAL 文件: ~2MB (10k 条 JSONL), 追加写, 每 5min 滚动归档
```

### ── RollingAggregator (滚动窗口聚合器) ──

```rust
/// 滚动窗口聚合器——每 5 分钟封存一个窗口, 保留最近 12 个窗口 (1h)
/// 解决: 预计算聚合器只增不归零的问题
/// 启动时从快照文件恢复, 崩溃最多丢 5 分钟

pub struct RollingAggregator {
    current: Mutex<AggregatorWindow>,                         // 当前活跃窗口 (进行中)
    completed: Mutex<VecDeque<AggregatorWindow>>,              // 已完成窗口 (最多 12 个 = 1h)
    snapshot_path: PathBuf,                                   // 快照路径 /tmp/zen-aggregator-snapshot.json
}

/// 单个 5 分钟窗口——5 个一维预计算 HashMap
struct AggregatorWindow {
    window_start: i64,   // unix millis, 整 5 分钟对齐
    by_model: HashMap<String, ModelCounter>,
    by_node: HashMap<String, NodeCounter>,
    by_pool: HashMap<String, PoolCounter>,
    by_status: HashMap<u16, StatusCounter>,
    by_client: HashMap<String, ClientCounter>,
}

struct ModelCounter {
    requests: u64,
    prompt_tokens: u64,
    completion_tokens: u64,
    total_tokens: u64,
    bytes_sent: u64,
    bytes_received: u64,
    count_429: u64,
    sum_latency_ms: f64,
    count_latency: u64,
}
// NodeCounter / PoolCounter / StatusCounter / ClientCounter 同 ModelCounter

impl RollingAggregator {
    /// 写入: ~1.5μs, 更新 5 个 HashMap + 一次窗口边界检查
    fn record(&self, tele: &RequestTelemetry) {
        let mut cur = self.current.lock();
        // 检查是否跨越 5 分钟边界
        if tele.ts - cur.window_start >= 300_000 {
            // 封存当前窗口 -> completed
            let closed = std::mem::replace(&mut *cur, AggregatorWindow::new(tele.ts));
            let mut c = self.completed.lock();
            c.push_back(closed);
            while c.len() > 12 { c.pop_front(); }  // 只保留最近 1h
            drop(c);
            // 异步写快照文件 (不阻塞当前请求)
            self.snapshot_background();
        }
        cur.update(tele);  // 更新 5 个 HashMap
    }

    /// 聚合查询: O(1) 读预计算维度
    fn query(
        &self,
        since: i64,
        until: i64,
        group: GroupBy,
        aggregate: Aggregate,
    ) -> AggregatedResult {
        // 1. 找出 since~until 范围内涉及的完成窗口 (completed)
        // 2. + 当前窗口中落在 since~until 的部分 (current)
        // 3. 对每个窗口, 读对应的预计算 HashMap (O(1))
        // 4. 如果 group 是多维组合 (如 model×pool):
        //    返回 fallback_to_scan = true, 调用方走 RingBuffer 扫描聚合
    }

    /// 快照恢复 (启动时调用)
    fn load_snapshot(&self) {
        let file = File::open(&self.snapshot_path).ok();
        if let Some(f) = file {
            let snap: Vec<AggregatorWindow> = serde_json::from_reader(f).ok().unwrap_or_default();
            if snap.is_empty() { tracing::warn!("aggregator snapshot corrupted, starting fresh"); }
            let mut c = self.completed.lock();
            // 只加载最近 24h 的快照
            let cutoff_24h = unix_ms() - 86_400_000;
            for w in snap.into_iter().filter(|w| w.window_start >= cutoff_24h) {
                c.push_back(w);
            }
        }
    }

    /// 快照持久化: 每 5 分钟原子写入 /tmp/zen-aggregator-snapshot.json
    /// 写临时文件 -> rename 覆盖, 避免崩溃产生损坏的半写文件
    fn snapshot_background(&self) {
        let c = self.completed.lock();
        let snap: Vec<&AggregatorWindow> = c.iter().collect();
        let tmp = self.snapshot_path.with_extension("snapshot.tmp");
        if let Ok(f) = File::create(&tmp) {
            serde_json::to_writer(f, &snap).ok();
            std::fs::rename(&tmp, &self.snapshot_path).ok();
        }
    }
}

/// 查询路径矩阵:
/// ┌─────────────────────┬──────────────────┬────────────────────────────┐
/// │ group               │ aggregate        │ 查询路径                   │
/// ├─────────────────────┼──────────────────┼────────────────────────────┤
/// │ model/node/pool/    │ tokens/bytes/    │ 预计算 HashMap -> O(1)      │
/// │ client/status       │ count/latency_avg│                            │
/// ├─────────────────────┼──────────────────┼────────────────────────────┤
/// │ model,pool (多维)   │ 同上             │ 标记 fallback -> 环形缓冲   │
/// │                     │                  │ 区扫描 ~5ms (10k 条)       │
/// ├─────────────────────┼──────────────────┼────────────────────────────┤
/// │ 原始明细 (no group)  │ —               │ 环形缓冲区扫描 ~2ms        │
/// └─────────────────────┴──────────────────┴────────────────────────────┘
/// 时间范围匹配:
///   since ~ until 在 1h 内 -> 聚合器窗口链精确拼接
///   since ~ until 超 1h   -> 拼接聚合器窗口 (1h) + 扫描环形缓冲区 (剩余部分)
///   since 超 24h           -> 返回截断结果 + "truncated": true
```

/// DefaultCollector——连接 RingBuffer + RollingAggregator + WAL + MetricsExport
pub struct DefaultCollector {
    ring_buffer: RingBuffer<RequestTelemetry>,  // 固定内存 2MB
    aggregator: RollingAggregator,              // 滚动窗口 + 快照
    wal: WAL,                                   // 写前日志
    export: MetricsExport,                      // Prometheus 导出

    // 原子计数器 (保留, 用于 /metrics 热路径)
    total_requests: AtomicU64,
    success_count: AtomicU64,
    count_429: AtomicU64,
    count_4xx: AtomicU64,
    count_5xx: AtomicU64,
    count_timeout: AtomicU64,
    bytes_sent: AtomicU64,
    bytes_received: AtomicU64,
    active_requests: AtomicI64,
    rpm_window: Mutex<VecDeque<Instant>>,
}

impl DefaultCollector {
    pub fn new(wal_path: PathBuf, snapshot_path: PathBuf) -> Self {
        let mut rb = RingBuffer::new(10_000);
        let wal = WAL::new(&wal_path);
        // 启动回放
        wal.refill_ring_buffer(&mut rb);
        let agg = RollingAggregator::new(&snapshot_path);
        agg.load_snapshot();
        Self { ring_buffer: rb, aggregator: agg, wal, ... }
    }

    /// 请求入口: 在 proxy.rs 每个请求完成后调用
    pub fn record_request(&self, tele: &RequestTelemetry) {
        // ① 原子计数器 (热路径, ~0.1μs)
        self.total_requests.fetch_add(1, Ordering::Relaxed);
        // status 分类计数器 ...
        // ② RPM 滑动窗口
        self.rpm_window.lock().push_back(Instant::now());
        // ③ 环形缓冲区 + WAL (~0.8μs)
        self.ring_buffer.push(tele);
        // ④ 滚动窗口聚合器 (~1.5μs)
        self.aggregator.record(tele);
    }

    /// 查询接口: 由 /admin/requests 处理函数调用
    pub fn query_requests(
        &self,
        since: i64, until: i64,
        filters: &[Filter],
        group: GroupBy,
        aggregate: Aggregate,
        limit: usize,
        cursor: Option<u64>,
    ) -> QueryResult {
        match (group, aggregate) {
            // 预计算维度 -> 走聚合器 O(1)
            (GroupBy::Model | GroupBy::Node | GroupBy::Pool | GroupBy::Status | GroupBy::Client, _) => {
                let agg_result = self.aggregator.query(since, until, group, aggregate);
                // 如果时间范围超出聚合器窗口 -> 环形缓冲区补充
                if agg_result.truncated {
                    let scan = self.ring_buffer.query(since, filters, limit, cursor);
                    QueryResult::Hybrid { agg: agg_result, scan }
                } else {
                    QueryResult::Aggregated(agg_result)
                }
            }
            // 多维组合 / 原始明细 -> 走环形缓冲区扫描
            _ => {
                let (items, cursor, truncated) = self.ring_buffer.query(since, filters, limit, cursor);
                QueryResult::Raw { items, cursor, truncated }
            }
        }
    }

    /// 事件查询: 由 /admin/events 处理函数调用
    pub fn query_events(&self, filters: &[EventFilter], limit: usize)
        -> Vec<PoolEvent> { ... }
}

/// 写前日志——追加写, 崩溃回放, 归档清理
impl DefaultCollector {

    pub fn rotate_wal(&self) { self.wal.rotate(); }
    /// 每 5 分钟后台任务
    pub fn snapshot_aggregator(&self) { self.aggregator.snapshot_background(); }
    /// 启动时 + 每 24h
    pub fn purge_wal(&self) { self.wal.purge_old(); }
}

    SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_millis() as i64
}

pub fn today_ymd() -> u32 {
    let days = unix_ms() / 86_400_000;
    ((days / 146097) as u32 * 10000) + 19700101
}

pub fn sha256_first8(input: &str) -> String {
    use sha2::{Sha256, Digest};
    let hash = Sha256::digest(input.as_bytes());
    hex::encode(&hash[..4])
}

pub fn redact_socks_url(url: &str) -> String {
    if let Some(rest) = url.strip_prefix("socks5://") {
        if let Some(at) = rest.find('@') {
            return format!("socks5://***@{}", &rest[at + 1..]);
        }
    }
    url.to_string()
}

```rust
pub async fn proxy_handler<P, C>(
    State(state): State<Arc<ProxyState<P, C>>>,
    method: Method,
    Path(path): Path<String>,
    headers: HeaderMap,
    body: Bytes,
) -> Response
where
    P: PoolManager,
    C: DataCollector,
{
    // ── 计时点 1: 请求开始 ──
    let t_start = Instant::now();

    // ── 模型名 + client_id 提取 ──
    let config = &state.config;
    let model = extract_model(&body);                          // 从 body["model"]
    let client_id = headers.get("authorization")               // SHA256[:8] of Bearer token
        .and_then(|v| v.to_str().ok())
        .and_then(|v| v.strip_prefix("Bearer "))
        .map(|t| sha256_first8(t));
    let body = apply_model_override(&body, config);            // 模型名映射
    let is_streaming = detect_streaming(&body);

    // ── 请求体大小 ──
    let bytes_sent = body.len() as u64;

    // ── 调度节点 ──
    let t_schedule = Instant::now();
    let dispatch_result = match state.pool_manager.dispatch(&RequestMeta::from(&body)) {
        Err(DispatchError::NoResource) => {
            // 无资源可用: 返回 503 + 记录 telemetry
            state.collector.record_request(&RequestTelemetry {
                rid: rid(), ts: unix_ms(),
                model, client_id,
                path: path.clone(), method: method.to_string(), is_streaming,
                node_url: String::new(), pool: String::new(), exit_ip: None,
                status: 503, rate_limited: false, retry_count: 0,
                latency_total_ms: t_start.elapsed().as_millis_f64(),
                upstream_ms: 0.0, ttft_ms: None,
                prompt_tokens: None, completion_tokens: None, total_tokens: None,
                bytes_sent, bytes_received: 0, tokens_per_kb: None,
            });
            return (StatusCode::SERVICE_UNAVAILABLE, json!({
                "error": { "code": -999, "message": "没有资源可用" }
            })).into_response();
        }
        Ok(r) => r,
    };
    let node_url = redact_socks_url(&dispatch_result.node.url().to_string());
    let pool = dispatch_result.node.pool().to_string();

    // ── 计时点 2: 上游请求开始 ──
    let t_upstream = Instant::now();

    // ── 代理转发 (含重试) ──
    let result = proxy_with_retry(
        &dispatch_result, &path, &method, &body, is_streaming,
    ).await;

    // ── 汇报结果给 PoolManager ──
    state.pool_manager.report(
        dispatch_result.node.id(),
        &result,
    );

    // ── 计时点 3: 结束 ──
    let latency_total_ms = t_start.elapsed().as_millis_f64();
    let upstream_ms = t_upstream.elapsed().as_millis_f64();

    // ── 组装 RequestTelemetry ──
    let (status, retry_count, bytes_received, prompt_tokens, completion_tokens, total_tokens, exit_ip, ttft_ms) =
        match &result {
            Ok(resp_meta) => (
                resp_meta.status, resp_meta.retry_count,
                resp_meta.bytes_received,
                resp_meta.prompt_tokens, resp_meta.completion_tokens, resp_meta.total_tokens,
                resp_meta.exit_ip.clone(), resp_meta.ttft_ms,
            ),
            Err((status, retry_count)) => (*status, *retry_count, 0, None, None, None, None, None),
        };

    let total_tokens_val = total_tokens.or(
        // 如果上游没返回 usage, 用 bytes 估算
        (bytes_received > 0).then(|| (bytes_received as f64 / 4.0) as u64)
    );
    let tokens_per_kb = total_tokens_val
        .zip((bytes_received > 0).then_some(bytes_received))
        .map(|(t, b)| t as f64 * 1024.0 / b as f64);

    state.collector.record_request(&RequestTelemetry {
        rid: rid(), ts: unix_ms(),
        model, client_id,
        path, method: method.to_string(), is_streaming,
        node_url, pool, exit_ip,
        status, rate_limited: false, retry_count: retry_count as u32,
        latency_total_ms, upstream_ms, ttft_ms,
        prompt_tokens, completion_tokens, total_tokens: total_tokens_val,
        bytes_sent, bytes_received, tokens_per_kb,
    });

    // ── 响应 ──
    match result {
        Ok(resp_meta) => resp_meta.response,
        Err((status, _)) => {
            (StatusCode::from_u16(status).unwrap(), json!({
                "error": { "message": format!("upstream error {}", status) }
            })).into_response()
        }
    }
}

```rust
/// 返回元数据, 不再包装 Response 为 Result<Response, u16>
/// 方便调用方提取 token / bytes / ttft 等字段
struct ProxyResult {
    response: Response,
    status: u16,
    retry_count: u32,
    bytes_received: u64,
    prompt_tokens: Option<u64>,
    completion_tokens: Option<u64>,
    total_tokens: Option<u64>,
    exit_ip: Option<String>,
    ttft_ms: Option<f64>,
}

/// 带重试的代理转发——内部紧耦合, 但所有可观察数据通过 ProxyResult 向外暴露
async fn proxy_with_retry(
    dispatch: &DispatchResult,
    path: &str,
    method: &Method,
    body: &[u8],
    is_streaming: bool,
) -> Result<ProxyResult, (u16, u32)> {
    let max_retries = 3;
    let mut last_status = 502u16;
    let mut retry_count = 0u32;

    for attempt in 0..=max_retries {
        let upstream = build_upstream_url(
            &dispatch.config.upstream_base, &format!("v1/{}", path),
        );

        let req = dispatch.client
            .request(reqwest::Method::POST, &upstream)
            .header("Content-Type", "application/json")
            .body(body.to_vec());

        let t_req = Instant::now();
        match req.send().await {
            Ok(up_resp) => {
                let status = up_resp.status().as_u16();
                if status < 400 {
                    let bytes_received;
                    let (prompt_tokens, completion_tokens, total_tokens);

                    if is_streaming {
                        // 流式: 逐 chunk 累加 bytes + 末 chunk 提取 usage
                        let (resp, b_recv, ttft, pt, ct, tt) = stream_to_axum(up_resp).await;
                        bytes_received = b_recv;
                        prompt_tokens = pt; completion_tokens = ct; total_tokens = tt;
                        return Ok(ProxyResult {
                            response: resp, status, retry_count,
                            bytes_received, prompt_tokens, completion_tokens, total_tokens,
                            exit_ip: None, ttft_ms: Some(ttft),
                        });
                    } else {
                        // 非流式: 全量 body -> 解析 usage
                        let (resp, b_recv, pt, ct, tt) = read_full_body(up_resp).await;
                        bytes_received = b_recv;
                        prompt_tokens = pt; completion_tokens = ct; total_tokens = tt;
                        return Ok(ProxyResult {
                            response: resp, status, retry_count,
                            bytes_received, prompt_tokens, completion_tokens, total_tokens,
                            exit_ip: None, ttft_ms: None,
                        });
                    }
                }
                last_status = status;
                retry_count = attempt + 1;
                if !should_retry(status, attempt, max_retries) {
                    return Err((status, retry_count));
                }
                tokio::time::sleep(Duration::from_secs_f64(
                    smart_backoff(attempt, Some(status))
                )).await;
            }
            Err(_) => {
                last_status = 502;
                retry_count = attempt + 1;
                if attempt < max_retries {
                    tokio::time::sleep(Duration::from_secs_f64(
                        smart_backoff(attempt, None)
                    )).await;
                }
            }
        }
    }
    Err((last_status, retry_count))
}

/// 非流式 body 读取 + usage 提取 (旁路, 不阻塞主路径)
async fn read_full_body(resp: Response) -> (Response, u64, Option<u64>, Option<u64>, Option<u64>) {
    let bytes = resp.bytes().await.unwrap_or_default();
    let len = bytes.len() as u64;
    let (pt, ct, tt) = if let Ok(v) = serde_json::from_slice::<Value>(&bytes) {
        if let Some(u) = v.get("usage") {
            ( u["prompt_tokens"].as_u64(),
              u["completion_tokens"].as_u64(),
              u["total_tokens"].as_u64() )
        } else { (None, None, None) }
    } else { (None, None, None) };
    // 保持与现有响应格式兼容
    let patched = patch_response_content(&bytes);
    ( (StatusCode::OK, Json(patched)).into_response(), len, pt, ct, tt )
}

/// 流式 SSE 处理 + bytes 累加 + 末 chunk usage 检查
async fn stream_to_axum(resp: Response) -> (Response, u64, f64, Option<u64>, Option<u64>, Option<u64>) {
    let mut stream = resp.bytes_stream();
    let t_first = Instant::now();
    let mut total_bytes = 0u64;
    let mut got_first = false;
    let mut ttft = 0.0f64;
    let mut last_chunk: Option<Vec<u8>> = None;
    let mut tx: Option<mpsc::UnboundedSender<Result<Bytes, axum::Error>>> = None;

    let (tx_internal, rx) = tokio::sync::mpsc::unbounded_channel();
    let stream_out = tokio_stream::wrappers::UnboundedReceiverStream::new(rx);
    let mut tx = tx_internal.clone();

    while let Some(chunk) = stream.next().await {
        match chunk {
            Ok(c) => {
                if !got_first {
                    ttft = t_first.elapsed().as_millis_f64();
                    got_first = true;
                }
                total_bytes += c.len() as u64;
                let patched = patch_sse_line(&c);
                last_chunk = Some(patched.clone());
                tx.send(Ok(Bytes::from(patched))).ok();
            }
            Err(e) => {
                tx.send(Err(e)).ok();
                break;
            }
        }
    }
    // 从末 chunk 尝试提取 usage
    let (pt, ct, tt) = if let Some(last) = &last_chunk {
        if let Ok(v) = serde_json::from_slice::<Value>(last) {
            if let Some(u) = v.get("usage") {
                ( u["prompt_tokens"].as_u64(),
                  u["completion_tokens"].as_u64(),
                  u["total_tokens"].as_u64() )
            } else { (None, None, None) }
        } else { (None, None, None) }
    } else { (None, None, None) };

    let body_stream = Body::from_stream(stream_out);
    (Response::new(body_stream), total_bytes, ttft, pt, ct, tt)
}
